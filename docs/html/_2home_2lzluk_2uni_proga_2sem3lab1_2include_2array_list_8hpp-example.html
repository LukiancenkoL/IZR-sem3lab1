template <typename T>
class ArrayList : public List<T> {
public:
ArrayList()
: m_data(new T[4])
, m_size(0)
, m_capacity(4) {
}
ArrayList(const size_t capacity)
: m_data(new T[capacity])
, m_size(0)
, m_capacity(capacity) {
}
void sort(const Sort<T>& sort) {
sort.sort(*this);
}
size_t length() const {
return m_size;
}
size_t capacity() const {
return m_capacity;
}
void push_back(const T& element) {
if (m_size >= m_capacity) {
reallocate();
}
m_data[m_size++] = element;
}
void push_front(const T& element) {
if (m_size >= m_capacity) {
reallocate();
}
std::memmove(m_data + 1, m_data, m_size * sizeof(T));
m_data[0] = element;
m_size++;
}
const T& get(const size_t index) const {
if (index >= length()) {
throw std::out_of_range("index is out of bounds");
}
return m_data[index];
}
const T& get_first() const {
if (is_empty()) {
throw std::out_of_range("List is empty");
}
return get(0);
}
const T& get_last() const {
if (is_empty()) {
throw std::out_of_range("List is empty");
}
return get(length() - 1);
}
T& get_mut(const size_t index) const {
if (index >= length()) {
throw std::out_of_range("index is out of bounds");
}
return m_data[index];
}
T pop_front() {
if (is_empty()) {
throw std::out_of_range("List is empty");
}
T first = get_first();
std::memmove(m_data, m_data + 1, length() * sizeof(T));
m_size--;
return first;
}
T pop_back() {
if (is_empty()) {
throw std::out_of_range("List is empty");
}
return m_data[--m_size];
}
void insert(const size_t index, const T& element) {
if (index >= length()) {
push_back(element);
return;
}
if (index == 0) {
push_front(element);
return;
}
if (m_size >= m_capacity) {
reallocate();
}
std::memmove(m_data + index + 1, m_data + index,
(length() - index) * sizeof(T));
m_data[index] = element;
m_size++;
}
T remove(const size_t index) {
if (index >= length()) {
throw std::out_of_range("index is out of bounds");
}
T tmp = get(index);
std::memmove(m_data + index, m_data + index + 1,
(length() - index - 1) * sizeof(T));
m_size--;
return tmp;
}
bool is_empty() const {
return m_size == 0;
}
~ArrayList() {
delete[] m_data;
m_data = nullptr;
m_capacity = 0;
m_size = 0;
}
private:
void reallocate() {
size_t new_capacity = m_capacity * 2;
if (new_capacity == 0) {
new_capacity = 4;
}
auto new_data = new T[new_capacity];
for (size_t i = 0; i < m_size; i++) {
new_data[i] = m_data[i];
}
delete[] m_data;
m_data = new_data;
m_capacity = new_capacity;
}
private:
T* m_data;
size_t m_size;
size_t m_capacity;
FRIEND_TEST(array_list, reallocate);
};
